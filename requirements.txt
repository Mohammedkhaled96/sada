# Windows desktop application.
# For the container, see docker/requirements.txt - it omits the Windows-only
# parts, which cannot run in a Linux image.

# Speech recognition. Reads OpenAI's own .pt checkpoints and carries its tiktoken
# vocabulary inside the package, so it needs nothing from huggingface.co.
openai-whisper==20250625

# Installed for its Silero VAD, which engine/chunking.py uses to split on
# silence. Its own CTranslate2 weights are not used - see docs/accuracy-notes.md.
faster-whisper>=1.2.0

# The microphone, for dictation. Ships its own PortAudio build, so there is
# nothing to install outside Python.
sounddevice>=0.5.0

numpy>=2.0

# Native Win32 controls, which is why NVDA and JAWS read the interface without
# any custom accessibility work. NVDA itself is written with wxPython.
wxPython>=4.3.0

# Word over COM, and the JAWS scripting API.
pywin32>=306

# Windows System Media Transport Controls, used to pause whatever is playing
# for the length of a dictation - the microphone hears the music too. Optional:
# app/mediapause.py reports itself unavailable without it and the setting is
# greyed out, so an older install keeps working untouched.
winsdk>=1.0.0b10

# torch is pulled in by openai-whisper. On a CPU-only machine install the smaller
# CPU wheel first to avoid downloading the CUDA build:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
